@app.cell
def _(HfApi, HfHubHTTPError, hf_token_form, mo, os):
    # 2b. APPLY / VERIFY — explicit token dependency for every Hub/tokenizer call.
    _token_value = ((hf_token_form.value or {}).get("token", "") or "").strip()
    token_source = "widget" if _token_value else "environment"
    if not _token_value:
        _token_value = os.environ.get("HF_TOKEN", "").strip()
    hf_download_token = _token_value or False
    if _token_value:
        try:
            _identity = HfApi(token=_token_value).whoami()
        except HfHubHTTPError:
            raise RuntimeError(
                "HF token gagal diverifikasi; cek token/akses lalu Apply ulang."
            ) from None
        _access_token = (_identity.get("auth") or {}).get("accessToken") or {}
        _token_role = _access_token.get("role")
        if not _token_role:
            _fine_grained = _access_token.get("fineGrained") or {}
            _scoped = _fine_grained.get("scoped") or []
            _has_write = any(
                any("write" in str(perm).lower() for perm in scope_item.get("permissions", []))
                for scope_item in _scoped
                if isinstance(scope_item, dict)
            )
            if _has_write:
                _token_role = "write"
            elif _fine_grained:
                _token_role = "fine-grained"
            else:
                _token_role = "unknown"
        auth_info = {
            "mode": "authenticated",
            "source": token_source,
            "username": _identity.get("name", "unknown"),
            "role": _token_role,
            "verified": True,
        }
        if _token_role in {"write", "admin"}:
            auth_message = (
                f"HF token terverifikasi (akses {_token_role}). Siap untuk unduh dan upload."
            )
        elif _token_role == "read":
            auth_message = (
                "HF token terverifikasi (akses read). Siap untuk unduh; gunakan token write bila"
                " HF_UPLOAD_ENABLED diaktifkan."
            )
        else:
            auth_message = "HF token terverifikasi. Siap digunakan oleh loader dan tokenizer."
    else:
        auth_info = {"mode": "anonymous", "source": "none", "role": "none", "verified": False}
        auth_message = (
            "Tanpa token: akses anonim untuk checkpoint publik. Isi token lalu klik Apply bila"
            " diperlukan."
        )
    mo.md("## 2b · Apply / verifikasi HF token\n" + auth_message)
    return (hf_download_token, auth_info)

@app.cell
def _(torch, F, mo):
    from dataclasses import dataclass, field

    # 4. BATCH & LOSS — synthetic IDs, shift sekali, prefix/padding label -100.
    @dataclass(frozen=True)
    class TokenizedRepairExample:
        """Prefix is already IT-formatted; target contains the closing target token."""

        instance_id: str
        encoder_ids: tuple[int, ...]
        decoder_prefix_ids: tuple[int, ...]
        target_ids: tuple[int, ...]
        metadata: dict = field(default_factory=dict)

    @dataclass
    class Seq2SeqBatch:
        encoder_input_ids: torch.Tensor
        encoder_attention_mask: torch.Tensor
        decoder_input_ids: torch.Tensor
        decoder_attention_mask: torch.Tensor
        labels: torch.Tensor

        def to(self, device: str | torch.device) -> "Seq2SeqBatch":
            return Seq2SeqBatch(
                self.encoder_input_ids.to(device),
                self.encoder_attention_mask.to(device),
                self.decoder_input_ids.to(device),
                self.decoder_attention_mask.to(device),
                self.labels.to(device),
            )

        @property
        def encoder_position_ids(self) -> torch.Tensor:
            return (self.encoder_attention_mask.long().cumsum(-1) - 1).clamp_min(0)

        @property
        def decoder_position_ids(self) -> torch.Tensor:
            return (self.decoder_attention_mask.long().cumsum(-1) - 1).clamp_min(0)

    @dataclass
    class ModelOutput:
        logits: torch.Tensor
        loss: torch.Tensor | None

    class RepairCollator:
        def __init__(self, pad_token_id: int = 0, ignore_index: int = -100):
            self.pad_token_id = pad_token_id
            self.ignore_index = ignore_index

        def __call__(self, examples: list[TokenizedRepairExample]) -> Seq2SeqBatch:
            if not examples:
                raise ValueError("Cannot collate an empty batch.")
            encoder_presence = [bool(example.encoder_ids) for example in examples]
            if any(encoder_presence) and not all(encoder_presence):
                raise ValueError(
                    "Do not mix decoder-only and encoder-decoder examples in one batch."
                )
            for example in examples:
                if not example.decoder_prefix_ids or not example.target_ids:
                    raise ValueError("Decoder prefix and target must each contain tokens.")
                if (
                    min((*example.encoder_ids, *example.decoder_prefix_ids, *example.target_ids))
                    < 0
                ):
                    raise ValueError("Input token IDs must be nonnegative.")
            enc_len = max(len(example.encoder_ids) for example in examples)
            dec_len = max(
                len(example.decoder_prefix_ids) + len(example.target_ids) - 1
                for example in examples
            )
            size = len(examples)
            encoder = torch.full((size, enc_len), self.pad_token_id, dtype=torch.long)
            decoder = torch.full((size, dec_len), self.pad_token_id, dtype=torch.long)
            enc_mask = torch.zeros_like(encoder, dtype=torch.bool)
            dec_mask = torch.zeros_like(decoder, dtype=torch.bool)
            labels = torch.full_like(decoder, self.ignore_index)
            for row, example in enumerate(examples):
                encoder[row, -len(example.encoder_ids) :] = torch.tensor(example.encoder_ids)
                enc_mask[row, -len(example.encoder_ids) :] = True
                combined = example.decoder_prefix_ids + example.target_ids
                inputs = combined[:-1]
                targets = list(combined[1:])
                targets[: len(example.decoder_prefix_ids) - 1] = [self.ignore_index] * (
                    len(example.decoder_prefix_ids) - 1
                )
                decoder[row, -len(inputs) :] = torch.tensor(inputs)
                dec_mask[row, -len(inputs) :] = True
                labels[row, -len(targets) :] = torch.tensor(targets)
            return Seq2SeqBatch(encoder, enc_mask, decoder, dec_mask, labels)

    def patch_cross_entropy(
        logits: torch.Tensor, labels: torch.Tensor, ignore_index: int = -100
    ) -> torch.Tensor:
        if logits.ndim != 3 or labels.ndim != 2 or tuple(logits.shape[:2]) != tuple(labels.shape):
            raise ValueError("Expected logits [B,L,V] and already aligned labels [B,L].")
        if not bool((labels != ignore_index).any()):
            return logits.sum() * 0
        return F.cross_entropy(
            logits.float().reshape(-1, logits.shape[-1]),
            labels.reshape(-1),
            ignore_index=ignore_index,
        )

    def synthetic_batch(spec, encoder_tokens, target_tokens, device):
        if not 2 <= encoder_tokens <= 128 or not 2 <= target_tokens <= 32:
            raise ValueError("Gunakan encoder_tokens 2..128 dan target_tokens 2..32 untuk smoke.")
        context = tuple([2] + [3 + index % 80 for index in range(encoder_tokens - 1)])
        target = tuple([18 + index % 60 for index in range(target_tokens - 1)] + [106])
        examples = [
            TokenizedRepairExample(
                "synthetic-1", context if spec.encoder else (), (2, 105, 7), target
            ),
            TokenizedRepairExample(
                "synthetic-2",
                context[:-1] if spec.encoder else (),
                (2, 105, 8),
                target[:-2] + (106,),
            ),
        ]
        batch = RepairCollator()(examples).to(device)
        first_target = 2
        assert batch.labels[0, first_target].item() == target[0]
        assert (batch.labels[0, :first_target] == -100).all()
        return batch

    mo.md(
        "## 4 · Batch dan patch loss\nIDs 0..127 bersifat synthetic; tokenizer asli tetap diuji terpisah."
    )
    return (
        TokenizedRepairExample,
        Seq2SeqBatch,
        ModelOutput,
        RepairCollator,
        patch_cross_entropy,
        synthetic_batch,
    )

    def transfer_text_checkpoint(backbone, path: Path, *, skip_words: bool = False) -> dict:
        prefix = "model.language_model."
        target = backbone.state_dict()
        expected = {
            key: value
            for key, value in target.items()
            if not (skip_words and key == "embed_tokens.weight")
        }
        with safe_open(str(path), framework="pt", device="cpu") as reader:
            available = set(reader.keys())
            missing = [prefix + key for key in expected if prefix + key not in available]
            if missing:
                raise ValueError(f"Missing checkpoint tensors: {missing[:5]}")
            for key, value in expected.items():
                shape = reader.get_slice(prefix + key).get_shape()
                if tuple(shape) != tuple(value.shape):
                    raise ValueError(
                        f"Shape mismatch for {key}: source {shape}, target {tuple(value.shape)}"
                    )
            if any(value.is_meta for value in expected.values()):
                raise ValueError("Initialize destination storage before checkpoint transfer.")
            ignored_shared = []
            skipped = {prefix + "embed_tokens.weight"} if skip_words else set()
            for source_key in available:
                if not source_key.startswith(prefix) or source_key in skipped:
                    continue
                key = source_key[len(prefix) :]
                if key in expected:
                    continue
                match = re.fullmatch(
                    r"layers\.(\d+)\.self_attn\.(k_proj|v_proj|k_norm|v_norm)\.weight", key
                )
                layers = getattr(backbone, "layers", ())
                if (
                    match
                    and int(match[1]) < len(layers)
                    and layers[int(match[1])].self_attn.is_kv_shared_layer
                ):
                    ignored_shared.append(key)
                else:
                    raise ValueError(f"Unmapped text tensor: {key}")
            with torch.no_grad():
                for key, destination in expected.items():
                    source = reader.get_tensor(prefix + key)
                    # Direct CPU-mmap -> destination copy avoids a second large GPU PLE tensor.
                    destination.copy_(source)
        return {
            "text_prefix": prefix,
            "tensors_copied": len(expected),
            "encoder_words_skipped": skip_words,
            "ignored_shared_kv_tensors": sorted(ignored_shared),
            "nontext_tensors_ignored": sum(not key.startswith(prefix) for key in available),
        }


    class ProbeLoRALinear(nn.Module):
        def __init__(self, base: nn.Linear, rank: int = 16):
            super().__init__()
            if rank <= 0:
                raise ValueError("LoRA rank must be positive.")
            self.base = base
            self.base.requires_grad_(False)
            self.rank = rank
            self.scaling = 2.0  # alpha = 2 * rank
            self.lora_A = nn.Parameter(
                torch.empty(rank, base.in_features, device=base.weight.device, dtype=torch.float32)
            )
            self.lora_B = nn.Parameter(
                torch.zeros(base.out_features, rank, device=base.weight.device, dtype=torch.float32)
            )
            nn.init.normal_(self.lora_A, std=0.02)

        def forward(self, values):
            delta = F.linear(F.linear(values.float(), self.lora_A), self.lora_B)
            return self.base(values) + delta.to(values.dtype) * self.scaling


    def configure_probe_training(model, *, policy: str = "lora", rank: int = 16) -> dict:
        if policy not in {"lora", "full"}:
            raise ValueError("Training policy must be lora or full.")
        model.requires_grad_(policy == "full")
        adapted = []
        if policy == "lora":
            for name, module in list(model.named_modules()):
                if isinstance(module, nn.Linear) and name.rsplit(".", 1)[-1] in {
                    "q_proj",
                    "k_proj",
                    "v_proj",
                    "o_proj",
                }:
                    parent_name, attribute = name.rsplit(".", 1)
                    setattr(
                        model.get_submodule(parent_name), attribute, ProbeLoRALinear(module, rank)
                    )
                    adapted.append(name)
            # Asymmetric alignment is new, so it must remain trainable too.
            projection = getattr(model, "encoder_projection", None)
            if isinstance(projection, nn.Linear):
                projection.requires_grad_(True)
        trainable = [parameter for parameter in model.parameters() if parameter.requires_grad]
        if not trainable:
            raise ValueError("No trainable parameters were selected.")
        return {
            "policy": policy,
            "rank": rank if policy == "lora" else None,
            "adapter_modules": adapted,
            "trainable_parameters": sum(parameter.numel() for parameter in trainable),
            "trainable_bytes": tensor_bytes(trainable),
        }


    def tensor_bytes(tensors) -> int:
        seen = set()
        total = 0
        for tensor in tensors:
            # Parameters are unique across the iterator; object identity also dedups optimizer aliases.
            if id(tensor) not in seen:
                seen.add(id(tensor))
                total += tensor.numel() * tensor.element_size()
        return total


    def gpu_memory(device) -> dict:
        if torch.device(device).type != "cuda":
            return dict(
                allocated_bytes=None,
                reserved_bytes=None,
                free_bytes=None,
                total_bytes=None,
                peak_allocated_bytes=None,
                peak_reserved_bytes=None,
            )
        free, total = torch.cuda.mem_get_info(device)
        return dict(
            allocated_bytes=torch.cuda.memory_allocated(device),
            reserved_bytes=torch.cuda.memory_reserved(device),
            free_bytes=free,
            total_bytes=total,
            peak_allocated_bytes=torch.cuda.max_memory_allocated(device),
            peak_reserved_bytes=torch.cuda.max_memory_reserved(device),
        )

            "guard_bytes": weights + trainable * 3 + int(headroom_gib * 2**30),

    def gpu_guard(estimate, device):

        if free < estimate["guard_bytes"]:
            return f"Free VRAM {free / 2**30:.2f} GiB < guard {estimate['guard_bytes'] / 2**30:.2f} GiB."
        return None

            {"role":"assistant","content":row["target_patch"]}]
        complete = decoder.apply_chat_template(messages,tokenize=False,add_generation_prompt=False,enable_thinking=False)
        assert isinstance(prefix,str) and isinstance(complete,str)
        prefix_ids = tuple(decoder.encode(prefix,add_special_tokens=False))
        full_ids = tuple(decoder.encode(complete.rstrip(),add_special_tokens=False))
        if full_ids[:len(prefix_ids)]!=prefix_ids:raise ValueError("prefix_boundary")
        target_ids = full_ids[len(prefix_ids):]
        if encoder_ids[0]!=2 or prefix_ids[0]!=2 or not target_ids or target_ids[-1]!=106:
            raise ValueError("special_token_policy")
        if len(encoder_ids)>encoder_limit or len(full_ids)-1>decoder_limit:raise ValueError("token_overflow")
        return TokenizedRepairExample(row["instance_id"],encoder_ids,prefix_ids,target_ids,metadata={"source":row.get("source","unknown")})


    def prepare_training_data(spec,local_file,cache_dir,encoder_limit,decoder_limit,seed,progress=None):
        if not 1<=encoder_limit<=131072 or not 1<=decoder_limit<=131072:
            raise ValueError("Invalid token limits.")
        source = "curated_pool"
        frozen_rows, pool_manifest = load_curated_pool(local_file)
        train_rows, dev_rows = frozen_rows["train"], frozen_rows["dev"]
        records = train_rows + dev_rows
        filters = {}
        assert spec.encoder is not None
        enc_ref=CHECKPOINTS[spec.encoder.size]["encoder"];dec_ref=CHECKPOINTS[spec.decoder.size]["decoder"]
        enc_assets=verified_assets(enc_ref,cache_dir);dec_assets=verified_assets(dec_ref,cache_dir)
        encoder=AutoTokenizer.from_pretrained(enc_assets["config.json"].parent,local_files_only=True,trust_remote_code=False)
        decoder=AutoTokenizer.from_pretrained(dec_assets["config.json"].parent,local_files_only=True,trust_remote_code=False)
        assert encoder is not None and decoder is not None
        token_filters=Counter()
        tokenized,kept = {},{}
        for split,rows in (("train",train_rows),("dev",dev_rows)):
            tokenized[split]=[];kept[split]=[]
            for index,row in enumerate(rows,1):
                if progress and (index == 1 or index % 250 == 0 or index == len(rows)): progress({"stage":"tokenization","message":f"{split}: {index:,}/{len(rows):,}","data_split":split,"items_done":index,"items_total":len(rows)})
                try:item=tokenize_editing_row(row,spec,encoder,decoder,encoder_limit,decoder_limit)
                except ValueError as error:
                    raise ValueError("Frozen pool rejected by tokenizer; check its revision and token limits.") from error
                tokenized[split].append(item);kept[split].append(row)
        if not all(tokenized.values()):raise ValueError("Empty split after token filtering.")
        manifest={"source":source,"source_ref":None,
            "split_policy":pool_manifest["report"]["split"]["policy"],
            "seed":seed,"encoder_limit":encoder_limit,"decoder_limit":decoder_limit,
            "tokenizer_sources":[enc_ref,dec_ref],"data_filters":filters,"token_filters":dict(token_filters),
            "train":editing_manifest(kept["train"]),"dev":editing_manifest(kept["dev"])}
        if pool_manifest is not None:
            manifest["curated_pool_fingerprint"]=pool_manifest["fingerprint"]
            manifest["split_policy"]=pool_manifest["report"]["split"]["policy"]
            manifest["pool_report"]=pool_manifest["report"]
        manifest["data_format_version"]=("repository_context_diff_v1" if all(r.get("record_type") == "repository_patch_v1" for r in records) else "paired_code_diff_v1")
        manifest["tokenized_fingerprint"]=stable_fingerprint({split:[(item.instance_id,item.encoder_ids,item.decoder_prefix_ids,item.target_ids) for item in items] for split,items in tokenized.items()})
        manifest["fingerprint"]=stable_fingerprint(manifest)
        return tokenized["train"],tokenized["dev"],manifest

    mo.md("## 8 · Data / tokenisasi\nSeluruh train/dev dari koleksi HF beku; overflow ditolak dan split tidak diubah.")
    return (editing_record, prepare_editing_records, split_editing_records, editing_manifest, stable_fingerprint, prepare_training_data,)

def _(

    HF_REPO_ID,

    experiment_spec,

    def test_family_policy_matches_selected_pair():
        spec = experiment_spec(EXPERIMENT)
        assert spec.tie_encoder_decoder_words == (FAMILY == "symmetric")
        assert EXPERIMENT in PAIR_OPTIONS[FAMILY]

    def test_repository_export_uses_public_context_and_rejects_eval():
        raw = {"record_type":"repository_patch_v1", "role":"train", "instance_id":"synthetic-export", "source":"bifocalgemma/synthetic-repository-fixture", "source_revision":"fixture-v1", "group":"fixture/repo", "commit":"a"*40, "repos":"fixture/repo", "instruction":"Fix helper.", "context":"### File: helper.py\nreturn 0\n", "target_patch":"diff --git a/helper.py b/helper.py\n", "snapshot_sha256":"b"*64, "context_selection":{"gold_used":False}}
        assert "pair_hash" not in raw and "instruction_hash" not in raw
        metadata = editing_manifest([raw])[0]
        assert metadata["instance_id"] == raw["instance_id"]
        assert "pair_hash" not in raw and "instruction_hash" not in raw
        row = editing_record(raw,source="editpackft")
        assert metadata["pair_hash"] == row["pair_hash"]
        assert metadata["instruction_hash"] == row["instruction_hash"]
        assert row["context"]==raw["context"]
        try:
            editing_record({**raw,"role":"eval"},source="editpackft")
            raise AssertionError("Eval export was admitted to training.")
        except ValueError as error:
            assert str(error)=="evaluation_record_cannot_enter_training"

    def test_editing_input_does_not_contain_target_code():
        row = editing_record(
            {
                "old_file": "a.py",
                "new_file": "a.py",
                "old_contents": "x = 0\n",
                "new_contents": "x = 1\n",
                "instruction": "Fix x",
                "commit": "c1",
            },
            source="editpackft",
        )
        assert row["before"] == "x = 0\n"
        assert "+x = 1" in row["target_patch"]
        assert row["instruction"] == "Fix x"

    def test_commit_groups_do_not_overlap():
        rows = []
        for i in range(6):
            rows.append(
                {
                    "old_file": "a.py",
                    "new_file": "a.py",
                    "old_contents": f"x = {i}\n",
                    "new_contents": f"x = {i + 10}\n",
                    "instruction": "Fix x",
                    "commit": f"c{i}",
                }
            )
        records, _ = prepare_editing_records(rows, source="editpackft")
        train, dev = split_editing_records(records, seed=17)
        assert not ({r["group"] for r in train} & {r["group"] for r in dev})

    def test_hub_upload_configuration():
        assert HF_REPO_ID == "daruokta/bifocalgemma-sft"
        path_in_repo = f"{FAMILY}/{EXPERIMENT}/{RUN_NAME}"
        assert path_in_repo.startswith(f"{FAMILY}/")
        assert FAMILY in {"symmetric", "asymmetric", "decoder_only"}
